```python
import asyncio
import time
from web3 import Web3
from loguru import logger

class FreeTierRPCManager:
    def __init__(self):
        self.endpoints = [CONFIG["rpc_primary"]] + CONFIG["rpc_fallbacks"]
        self.ws_endpoints = [CONFIG["rpc_ws_primary"]] + CONFIG["rpc_ws_fallbacks"]
        self.current_idx = 0
        self.latency_history = []

    async def get_w3(self) -> Web3:
        for _ in range(len(self.endpoints)):
            url = self.endpoints[self.current_idx]
            start = time.time()
            try:
                w3 = Web3(Web3.AsyncHTTPProvider(url))
                await w3.eth.block_number  # health check
                latency = (time.time() - start) * 1000
                self.latency_history.append(latency)
                logger.info(f"RPC {url} latency {latency:.1f}ms")
                return w3
            except:
                self.current_idx = (self.current_idx + 1) % len(self.endpoints)
        raise Exception("All free RPCs down")
```
